## Create a Service Bus

# Create a resource group

SERVICE_BUS_NAMESPACE=keda-demo-is-rubicon
QUEUE_NAME=demo-queue

# Create a Service Bus namespace
az servicebus namespace create \
    --resource-group $CLUSTER_RG \
    --name $SERVICE_BUS_NAMESPACE \
    --location $LOCATION \
    --sku Standard

# Create a Service Bus queue
az servicebus queue create \
    --resource-group $CLUSTER_RG \
    --namespace-name $SERVICE_BUS_NAMESPACE \
    --name $QUEUE_NAME \
    --enable-partitioning true \
    --default-message-time-to-live PT1M # Set the default message time to live to 1 minute

# Create a Service Bus queue authorization rule
az servicebus queue authorization-rule create \
    --resource-group $CLUSTER_RG \
    --namespace-name $SERVICE_BUS_NAMESPACE \
    --queue-name $QUEUE_NAME \
    --name queuerule \
    --rights Manage Listen Send

# Get the connection string for the Service Bus queue
SERVICEBUS_CONNECTIONSTRING=$(az servicebus queue authorization-rule keys list \
    --resource-group $CLUSTER_RG \
    --namespace-name $SERVICE_BUS_NAMESPACE \
    --queue-name $QUEUE_NAME \
    --name queuerule \
    --query primaryConnectionString -o tsv)

# Create a Kubernetes secret for the Service Bus connection string
kubectl create secret generic servicebus-connectionstring \
    --from-literal=connection=$SERVICEBUS_CONNECTIONSTRING \
    --namespace default

# Create a TriggerAuthentication for the Service Bus connection string
cat <<EOF | kubectl apply -f -
apiVersion: keda.sh/v1alpha1
kind: TriggerAuthentication
metadata:
  name: azure-servicebus-auth
spec: 
  secretTargetRef:
    - key: connection
      name: servicebus-connectionstring # name of secret
      parameter: connection # key in secret
EOF

# Create a Kubernetes Deployment for the consumer app
cat <<EOF | kubectl apply -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: consumer-app
spec:
  replicas: 1
  selector:
    matchLabels:
      app: consumer-app
  template:
    metadata:
      labels:
        app: consumer-app
    spec:
      containers:
      - name: consumer-app
        image: ams0/consumer-app:0.9
        env:
        - name: SERVICEBUS_CONNECTION_STRING
          valueFrom:
            secretKeyRef:
              name: servicebus-connectionstring
              key: connection
        - name: SERVICEBUS_QUEUE_NAME
          value: "demo-queue"
EOF

kubectl apply -f - <<EOF
# Create a ScaledObject for the Service Bus queue
apiVersion: keda.sh/v1alpha1
kind: ScaledObject
metadata:
  name: azure-servicebus-queue-scaledobject
  namespace: default
spec:
  scaleTargetRef:
    kind: Deployment
    name: consumer-app
  pollingInterval: 5
  cooldownPeriod: 10
  idleReplicaCount: 0
  minReplicaCount: 0
  maxReplicaCount: 10
  advanced:
    horizontalPodAutoscalerConfig:                  
      behavior:                    
        scaleDown:
          stabilizationWindowSeconds: 3
  triggers:
  - type: azure-servicebus
    metadata:
      queueName: $QUEUE_NAME
      messageCount: "5"
    authenticationRef:
        # reference to TriggerAuthentication
        name: azure-servicebus-auth
EOF
